Contact Kernel Factorization in Age and Immunity

Goal

This notebook builds illustrative ingredients for a factorized transmission surface

\[ \beta(\alpha,\gamma,\alpha',\gamma') = \tau\,B_{\alpha}(\alpha,\alpha')\,B_{\gamma}(\gamma,\gamma'), \] with \[ B_{\alpha}(\alpha,\alpha') = s(\alpha)K_{\alpha}(\alpha,\alpha')i(\alpha'), \qquad B_{\gamma}(\gamma,\gamma') = r(\gamma)K_{\gamma}(\gamma,\gamma')q(\gamma'). \]

Here \(\alpha\) is age and \(\gamma\) is immune status.

Helper Functions and Grids

rescale01 <- function(x) {
  rng <- range(x, na.rm = TRUE)
  if (diff(rng) == 0) return(rep(0, length(x)))
  (x - rng[1]) / diff(rng)
}

bump <- function(x, mu, sigma, height = 1) {
  height * exp(-0.5 * ((x - mu) / sigma)^2)
}

sigmoid_down <- function(x, mid, slope, low = 0, high = 1) {
  low + (high - low) / (1 + exp(slope * (x - mid)))
}

ages <- seq(0, 85, by = 1)
immune <- seq(0, 1, length.out = 101)

age_grid <- expand.grid(alpha = ages, alpha_p = ages) |>
  as_tibble()
immune_grid <- expand.grid(gamma = immune, gamma_p = immune) |>
  as_tibble()

Alternative Age-Side Ingredients

make_age_susceptibility <- function() {
  tibble(
    alpha = rep(ages, 3),
    value = c(
      rescale01(0.30 + bump(ages, 8, 9, 1.0) + bump(ages, 72, 14, 0.25)),
      rescale01(0.28 + bump(ages, 14, 10, 0.55) + bump(ages, 42, 18, 0.35)),
      rescale01(0.55 + bump(ages, 10, 14, 0.20) - bump(ages, 55, 18, 0.08))
    ),
    choice = factor(
      rep(c("Child-amplified", "Bimodal exposure", "Nearly flat"), each = length(ages)),
      levels = c("Child-amplified", "Bimodal exposure", "Nearly flat")
    )
  )
}

make_age_infectiousness <- function() {
  tibble(
    alpha_p = rep(ages, 3),
    value = c(
      rescale01(0.20 + bump(ages, 11, 7, 1.1) + bump(ages, 38, 16, 0.10)),
      rescale01(0.22 + bump(ages, 28, 11, 0.85) + bump(ages, 64, 16, 0.18)),
      rescale01(0.35 + bump(ages, 16, 13, 0.35) + bump(ages, 48, 18, 0.20))
    ),
    choice = factor(
      rep(c("School-age source", "Young-adult bridge", "Broad mild shedding"), each = length(ages)),
      levels = c("School-age source", "Young-adult bridge", "Broad mild shedding")
    )
  )
}

make_age_kernels <- function() {
  age_grid |>
    mutate(
      diag_band = exp(-((alpha - alpha_p)^2) / (2 * 8^2)),
      child_child = bump(alpha, 10, 6) * bump(alpha_p, 10, 6, 1.3),
      parent_child = bump(alpha, 35, 10) * bump(alpha_p, 9, 5, 0.9),
      child_parent = bump(alpha, 9, 5) * bump(alpha_p, 35, 10, 0.75),
      elder_care = bump(alpha, 75, 8) * bump(alpha_p, 48, 11, 0.6),
      school_hub = bump(alpha, 15, 7) * bump(alpha_p, 15, 7, 1.1),
      assortative = rescale01(1.00 * diag_band + 0.25 * parent_child + 0.20 * child_parent + 0.12),
      intergenerational = rescale01(0.55 * diag_band + 0.80 * parent_child + 0.65 * child_parent + 0.55 * elder_care + 0.08),
      school_centred = rescale01(0.45 * diag_band + 0.95 * school_hub + 0.30 * parent_child + 0.18)
    ) |>
    select(alpha, alpha_p, assortative, intergenerational, school_centred) |>
    pivot_longer(cols = c(assortative, intergenerational, school_centred), names_to = "choice", values_to = "K") |>
    mutate(choice = factor(choice,
      levels = c("assortative", "intergenerational", "school_centred"),
      labels = c("Assortative", "Intergenerational", "School-centred")
    ))
}

age_s_df <- make_age_susceptibility()
age_i_df <- make_age_infectiousness()
age_K_df <- make_age_kernels()

Alternative Immunity-Side Ingredients

make_immune_susceptibility <- function() {
  tibble(
    gamma = rep(immune, 3),
    value = c(
      1 - 0.75 * immune,
      sigmoid_down(immune, mid = 0.35, slope = 9, low = 0.08, high = 1),
      0.12 + 0.88 * exp(-4.2 * immune)
    ),
    choice = factor(
      rep(c("Leaky linear", "Threshold protection", "Near-sterilizing"), each = length(immune)),
      levels = c("Leaky linear", "Threshold protection", "Near-sterilizing")
    )
  )
}

make_immune_infectiousness <- function() {
  tibble(
    gamma_p = rep(immune, 3),
    value = c(
      0.25 + 0.75 * exp(-3.0 * immune),
      sigmoid_down(immune, mid = 0.25, slope = 11, low = 0.02, high = 1),
      0.15 + 0.85 * exp(-1.2 * immune)
    ),
    choice = factor(
      rep(c("Residual shedding", "Sharp sterilization", "Slow decay"), each = length(immune)),
      levels = c("Residual shedding", "Sharp sterilization", "Slow decay")
    )
  )
}

make_immune_kernels <- function() {
  immune_grid |>
    mutate(
      diag_band = exp(-((gamma - gamma_p)^2) / (2 * 0.12^2)),
      low_low = bump(gamma, 0.10, 0.12) * bump(gamma_p, 0.10, 0.12, 1.1),
      high_source_low_target = bump(gamma, 0.18, 0.16) * bump(gamma_p, 0.80, 0.14, 0.85),
      broad = rescale01(0.50 + 0.15 * diag_band),
      assortative = rescale01(0.20 + 1.00 * diag_band),
      low_hotspot = rescale01(0.10 + 0.85 * low_low + 0.55 * high_source_low_target + 0.10 * diag_band)
    ) |>
    select(gamma, gamma_p, broad, assortative, low_hotspot) |>
    pivot_longer(cols = c(broad, assortative, low_hotspot), names_to = "choice", values_to = "K") |>
    mutate(choice = factor(choice,
      levels = c("broad", "assortative", "low_hotspot"),
      labels = c("Broad", "Assortative", "Low-immunity hotspot")
    ))
}

One Concrete Factorization Choice

selected_s <- age_s_df |> filter(choice == "Child-amplified") |> transmute(alpha, value = value)
selected_i <- age_i_df |> filter(choice == "School-age source") |> transmute(alpha_p, value = value)
selected_Ka <- age_K_df |> filter(choice == "Intergenerational") |> select(alpha, alpha_p, K)
selected_r <- imm_r_df |> filter(choice == "Threshold protection") |> transmute(gamma, value = value)
selected_q <- imm_q_df |> filter(choice == "Residual shedding") |> transmute(gamma_p, value = value)
selected_Kg <- imm_K_df |> filter(choice == "Low-immunity hotspot") |> select(gamma, gamma_p, K)

B_alpha_df <- selected_Ka |>
  left_join(selected_s, by = "alpha") |>
  left_join(selected_i, by = "alpha_p") |>
  mutate(B = value.x * K * value.y)

B_gamma_df <- selected_Kg |>
  left_join(selected_r, by = "gamma") |>
  left_join(selected_q, by = "gamma_p") |>
  mutate(B = value.x * K * value.y)

B_alpha_plot <- ggplot(B_alpha_df, aes(alpha, alpha_p, fill = B)) +
  geom_raster() +
  coord_equal(expand = FALSE) +
  scale_fill_gradientn(colours = c("#fff5f0", "#fcbba1", "#fb6a4a", "#cb181d", "#67000d")) +
  labs(title = expression(B[alpha](alpha, alpha*"'") == s(alpha) %.% K[alpha](alpha, alpha*"'") %.% i(alpha*"'")), x = expression(alpha), y = expression(alpha*"'")) +
  theme_minimal(base_size = 12) +
  theme(panel.grid = element_blank())

B_gamma_plot <- ggplot(B_gamma_df, aes(gamma, gamma_p, fill = B)) +
